#!/usr/bin/env python3
"""Export one experiment's evaluation pool and results as a publishable dataset.

Reads (under --root, the folder that holds manifests/ and runs/):
  manifests/<experiment>/pool-index.json, manifests/<experiment>/<protocol>/<task>.json
  runs/<experiment>/suite.json, runs/<experiment>/<model>/<protocol>/<task>/{run.json,episode-*.result.json}
  runs/<experiment>/tables/*            (if tools/experiment.py has written them)

Writes <out>/ and <out>.zip:
  README.md                             what is inside and how to use it
  summary.md                            per-task success rates per model
  provenance.json                       runtime fingerprint, pool index hash, suite, policy identities
  pool/seeds.json                       accepted seeds in episode order and rejected seeds with reasons, per task
  pool/conditions.csv                   one row per accepted episode: seed, hashes, the 11 sampled attributes, instruction
  results/<model>/episodes.csv          one row per evaluated episode: success, actions, seconds, status
  results/<model>/identity.json         policy identity (name, config/adapter hashes) and manifest hashes it was pinned to
  tables/                               copy of runs/<experiment>/tables
  full/manifests/<protocol>/<task>.json the sealed manifests verbatim (initial states, expert info, provenance)
  full/pool-index.json, full/suite.json

Standard library only; run it on the machine that holds the data:
  python3 export_release.py --root ~/robotwin-data --experiment phys-pro6000-100-v6
Verify a copy with analysis/verify_release.py (recomputes every condition hash from the seeds).
"""
from __future__ import annotations

import argparse
import csv
from collections import defaultdict
import datetime as dt
import hashlib
import json
from pathlib import Path
import shutil

COLUMNS = ['task', 'episode_index', 'seed', 'condition_sha256', 'initial_state_sha256', 'protocol', 'active_factors',
           'mass_scale', 'com_offset_x_m', 'com_offset_y_m', 'com_offset_z_m', 'geometry_scale', 'table_tilt_deg',
           'table_tilt_azimuth_deg', 'table_height_offset_m', 'force_mg', 'force_direction_x', 'force_direction_y',
           'force_direction_z', 'camera_mode', 'camera_direction_x', 'camera_direction_y', 'camera_direction_z',
           'camera_orbit_m', 'camera_translation_m', 'camera_rotation_deg', 'friction', 'restitution',
           'object_joint_damping', 'arm_damping_scale', 'gripper_damping_scale', 'expert_seconds', 'instruction']


def flatten(task, ep):
    c = ep['condition']
    cam = c['camera']
    row = dict(task=task, episode_index=ep['episode_index'], seed=ep['seed'], condition_sha256=ep['condition_sha256'],
               initial_state_sha256=ep.get('initial_state_sha256', ''), protocol=c['protocol'],
               active_factors='|'.join(c['active_factors']), mass_scale=c['mass_scale'],
               com_offset_x_m=c['com_offset_m'][0], com_offset_y_m=c['com_offset_m'][1], com_offset_z_m=c['com_offset_m'][2],
               geometry_scale=c['geometry_scale'], table_tilt_deg=c['table_tilt_deg'],
               table_tilt_azimuth_deg=c['table_tilt_azimuth_deg'], table_height_offset_m=c['table_height_offset_m'],
               force_mg=c['force_mg'], force_direction_x=c['force_direction'][0], force_direction_y=c['force_direction'][1],
               force_direction_z=c['force_direction'][2], camera_mode=cam['mode'], camera_direction_x=cam['direction'][0],
               camera_direction_y=cam['direction'][1], camera_direction_z=cam['direction'][2], camera_orbit_m=cam['orbit_m'],
               camera_translation_m=cam['translation_m'], camera_rotation_deg=cam['rotation_deg'], friction=c['friction'],
               restitution=c['restitution'], object_joint_damping=c['object_joint_damping'],
               arm_damping_scale=c['arm_damping_scale'], gripper_damping_scale=c['gripper_damping_scale'],
               expert_seconds=ep.get('expert_seconds', ''), instruction=ep.get('instruction', ''))
    return {k: ('' if row[k] is None else row[k]) for k in COLUMNS}


def write_csv(path, rows, columns):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('w', newline='') as f:
        w = csv.DictWriter(f, fieldnames=columns)
        w.writeheader()
        w.writerows(rows)


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def export(root, experiment, out, protocol, models=None):
    root, out = Path(root).expanduser(), Path(out).expanduser()
    manifests = root / 'manifests' / experiment
    runs = root / 'runs' / experiment
    if out.exists():
        raise FileExistsError(f'Output exists: {out}')
    suite = json.loads((runs / 'suite.json').read_text()) if (runs / 'suite.json').exists() else None
    index = json.loads((manifests / 'pool-index.json').read_text()) if (manifests / 'pool-index.json').exists() else None
    tasks = sorted(p.stem for p in (manifests / protocol).glob('*.json') if not p.name.endswith('.progress.json'))
    if not tasks:
        raise FileNotFoundError(f'No manifests under {manifests / protocol}')
    (out / 'full/manifests' / protocol).mkdir(parents=True)
    seeds, rows, provenances, counts = {}, [], {}, {}
    for task in tasks:
        path = manifests / protocol / f'{task}.json'
        m = json.loads(path.read_text())
        if m.get('status') != 'verified':
            raise ValueError(f'{task}: manifest status {m.get("status")}')
        shutil.copy2(path, out / 'full/manifests' / protocol / path.name)
        seeds[task] = dict(start_seed=m['start_seed'], accepted=[e['seed'] for e in m['episodes']],
                           rejected={str(r['seed']): r.get('reason', r.get('status')) for r in m['rejected']},
                           manifest_sha256=m['sha256'])
        rows.extend(flatten(task, e) for e in m['episodes'])
        provenances[json.dumps(m['provenance'], sort_keys=True)] = m['provenance']
        counts[task] = dict(accepted=len(m['episodes']), rejected=len(m['rejected']),
                            candidates=len(m['episodes']) + len(m['rejected']))
    for name in ('pool-index.json',):
        if (manifests / name).exists():
            shutil.copy2(manifests / name, out / 'full' / name)
    if suite is not None:
        shutil.copy2(runs / 'suite.json', out / 'full/suite.json')
    (out / 'pool').mkdir()
    (out / 'pool/seeds.json').write_text(json.dumps(dict(experiment=experiment, protocol=protocol, tasks=seeds), indent=1) + '\n')
    write_csv(out / 'pool/conditions.csv', rows, COLUMNS)
    by_key = {(r['task'], r['episode_index']): r for r in rows}

    results, identities = {}, {}
    model_dirs = [p for p in sorted(runs.iterdir()) if p.is_dir() and (p / protocol).exists()] if runs.exists() else []
    for model_dir in model_dirs:
        model = model_dir.name
        if models and model not in models:
            continue
        rows_m, pinned, identity = [], {}, None
        for task_dir in sorted((model_dir / protocol).iterdir()):
            if not task_dir.is_dir() or task_dir.name not in seeds:
                continue
            run = task_dir / 'run.json'
            if run.exists():
                meta = json.loads(run.read_text())
                pinned[task_dir.name] = meta.get('manifest_sha256')
                identity = identity or meta.get('policy')
                if identity != meta.get('policy'):
                    raise ValueError(f'{model}: policy identity differs between tasks ({task_dir.name})')
            for res in sorted(task_dir.glob('episode-*.result.json')):
                r = json.loads(res.read_text())
                idx = int(res.name.split('-')[1].split('.')[0])
                pool = by_key.get((task_dir.name, idx))
                if r.get('status') == 'completed' and pool and r.get('condition_sha256') not in (None, pool['condition_sha256']):
                    raise ValueError(f'{model}/{task_dir.name}/{idx}: result condition hash differs from the manifest')
                rows_m.append(dict(task=task_dir.name, episode_index=idx, seed=pool['seed'] if pool else '',
                                   status=r.get('status'), success=('' if r.get('status') != 'completed' else int(bool(r.get('success')))),
                                   actions=r.get('actions', ''), seconds=round(r['seconds'], 1) if isinstance(r.get('seconds'), (int, float)) else '',
                                   benchmark_eligible=r.get('benchmark_eligible', '')))
        if not rows_m:
            continue
        write_csv(out / 'results' / model / 'episodes.csv', rows_m,
                  ['task', 'episode_index', 'seed', 'status', 'success', 'actions', 'seconds', 'benchmark_eligible'])
        (out / 'results' / model / 'identity.json').write_text(json.dumps(dict(policy=identity, manifests=pinned), indent=1) + '\n')
        identities[model] = identity
        per_task = defaultdict(lambda: [0, 0])
        for r in rows_m:
            if r['status'] == 'completed':
                per_task[r['task']][0] += r['success']
                per_task[r['task']][1] += 1
        results[model] = {t: dict(successes=s, episodes=n, rate=s / n) for t, (s, n) in per_task.items()}
    if (runs / 'tables').exists():
        shutil.copytree(runs / 'tables', out / 'tables')

    provenance = dict(experiment=experiment, protocol=protocol, exported=dt.datetime.now(dt.timezone.utc).isoformat(timespec='seconds'),
                      suite=suite, pool_index_sha256=index['sha256'] if index else None,
                      runtime_fingerprint=list(provenances.values()) if len(provenances) != 1 else next(iter(provenances.values())),
                      runtime_fingerprint_is_uniform=len(provenances) == 1, pool_counts=counts, policy_identities=identities)
    (out / 'provenance.json').write_text(json.dumps(provenance, indent=1) + '\n')

    lines = [f'# {experiment} / {protocol}', '', f'{len(tasks)} tasks; accepted episodes per task: '
             + ', '.join(sorted({str(c["accepted"]) for c in counts.values()})) + '.', '']
    if results:
        model_names = sorted(results)
        lines.append('| task | ' + ' | '.join(model_names) + ' |')
        lines.append('|---|' + '---:|' * len(model_names))
        for task in tasks:
            cells = []
            for m in model_names:
                cell = results[m].get(task)
                cells.append(f'{100 * cell["rate"]:.0f}% ({cell["successes"]}/{cell["episodes"]})' if cell else '-')
            lines.append(f'| {task} | ' + ' | '.join(cells) + ' |')
        means = []
        for m in model_names:
            complete = [v['rate'] for v in results[m].values() if v['episodes'] == counts[next(iter(counts))]['accepted']]
            means.append(f'{100 * sum(complete) / len(complete):.2f}% over {len(complete)} complete tasks' if complete else '-')
        lines.append('| **mean of task rates** | ' + ' | '.join(means) + ' |')
    else:
        lines.append('No policy results in this export.')
    (out / 'summary.md').write_text('\n'.join(lines) + '\n')
    (out / 'README.md').write_text(README.format(experiment=experiment, protocol=protocol, tasks=len(tasks)))
    archive = shutil.make_archive(str(out), 'zip', root_dir=out.parent, base_dir=out.name)
    size = Path(archive).stat().st_size
    print(f'{out}  ({sum(p.stat().st_size for p in out.rglob("*") if p.is_file()) / 1e6:.1f} MB unpacked)')
    print(f'{archive}  ({size / 1e6:.1f} MB, sha256 {sha(Path(archive))})')
    return out


README = '''# RoboTwin-Phys reproduction dataset: {experiment} ({protocol})

Independent reimplementation of RoboTwin-Phys (arXiv 2609.26292); not the authors' official pool.
{tasks} tasks, expert-filtered episodes per task, shared by every evaluated policy.

## Files

- `pool/seeds.json`: for each task, the start seed, the accepted seeds in episode order, and every
  rejected seed with its reason (`expert_failed`, `unstable_initialization`). Scanning seeds upward from
  the start seed and keeping the accepted ones reproduces the pool.
- `pool/conditions.csv`: one row per accepted episode with the 11 sampled physical attributes, the
  condition hash, the initial-state hash and the language instruction. Conditions are a pure function of
  (task, seed): `robotwin_phys.conditions.sample(task, seed)` on the pinned repository recomputes them, and
  `analysis/verify_release.py` checks every row against that function.
- `results/<model>/episodes.csv`: per-episode policy outcome (success, actions taken, wall seconds).
  `identity.json` pins the policy name, profile and adapter hashes, and the manifest hashes used.
- `tables/`: aggregate tables as written by `tools/experiment.py` (if present).
- `full/`: the sealed manifests verbatim (sampled condition, verified initial scene state, expert
  result, runtime fingerprint), the pool index and the experiment suite. `provenance.json` summarises them.

## Using the pool elsewhere

`robotwin-phys evaluate` refuses a manifest whose runtime fingerprint (source, assets, package versions,
GPU, driver) differs from the current machine, so these manifests cannot be fed directly to a different
installation. On another machine, freeze the pool from the same start seed (`robotwin-phys freeze --seed
<start_seed>` or `tools/parallel_eval.py`); the expert is deterministic per seed on a given GPU class, so
the accepted seed list should match `pool/seeds.json`, and any difference is visible seed by seed.
'''


def main(argv=None):
    p = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    p.add_argument('--root', required=True, help='folder holding manifests/ and runs/')
    p.add_argument('--experiment', required=True)
    p.add_argument('--protocol', default='physical_random')
    p.add_argument('--models', nargs='*')
    p.add_argument('--out', help='default: <root>/release/<experiment>')
    a = p.parse_args(argv)
    return export(a.root, a.experiment, a.out or str(Path(a.root).expanduser() / 'release' / a.experiment), a.protocol, a.models)


if __name__ == '__main__':
    main()
ROBOTWIN_EXPORT
